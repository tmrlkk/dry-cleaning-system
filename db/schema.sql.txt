Client        (client_id PK, fio, phone UNIQUE, email)
Service       (service_id PK, name, description, base_price, hours)
Item          (item_id PK, kind, material, color, defects)
Status        (status_id PK, name)

OrderService  (order_id FK→Orders, service_id FK→Service, qty)
